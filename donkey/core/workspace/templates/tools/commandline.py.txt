"""执行命令的 `@tool` —— donkey 工作区内置工具。

本模块属于**材料层**（工作区），由 `donkey init` 生成，可由用户编辑或删除。

本模块**刻意不使用**持久 shell 会话：持久会话与将来的中断式交互相冲突。
每次调用都通过 ``subprocess.run`` **新建一个进程** —— 它天然可被中断、也
天然受超时约束。

已知缺口
--------
命令造成的**文件副作用不进变更记录** —— 命令绕开了工具的写路径，引擎层
看不到这次变更。引擎层将来靠**工作区快照**与变更记录对账。本模块**不**
试图拦截这类副作用（那属引擎层职责）。
"""

from __future__ import annotations

import json
import shlex
import subprocess

from langchain_core.tools import tool

try:  # 常规情形：作为工作区 ``tools`` 包的一部分被导入
    from .changes import resolve_path, workspace_root
except ImportError:  # pragma: no cover - 以顶层模块形态加载时的回退路径
    from changes import resolve_path, workspace_root  # type: ignore[no-redef]

#: 单次返回的**行数**上限（与读取侧同一组数值）。
MAX_OUTPUT_LINES = 2000

#: 单次返回的**字节**上限（50 KiB，与读取侧同一组数值）。
MAX_OUTPUT_BYTES = 50 * 1024


@tool
def run_shell(command: str, timeout: int = 60, cwd: str | None = None) -> str:
    """在新进程里执行一条命令，返回它的输出与退出码（JSON 文本）。

    每次调用都**新建一个进程**（没有持久 shell 会话）。命令行按 POSIX 引号
    规则被拆成参数数组，再以 ``shell=False`` **不经命令解释器**直接执行 ——
    命令解释器被完全绕开，从源头避免命令注入。命令运行超过 ``timeout``
    秒即被掐断，并以 ``timed_out`` 标记如实回报（**不抛异常**）。退出码
    ``0`` 视为成功，非零退出码原样出现在返回值里。

    工作目录默认是**工作区根**（绝不是进程当前目录），因此命令里的相对
    路径指向工作区内的文件。``cwd`` 给相对值时同样以工作区根为基准。

    Args:
        command: 要执行的命令行（如 ``"ls -la"``）。以**单个字符串**传入 ——
            工具内部按 POSIX 引号规则把它拆成参数数组。
        timeout: 最长执行秒数（默认 ``60``）。**不设上限** —— 调用方按场景
            自定；取值过大不会报错。
        cwd: 工作目录（绝对路径，或相对于工作区根的相对路径）。默认：工作区根。

    Returns:
        JSON 文本，含 ``stdout``（标准输出与标准错误**合并**后的文本）/
        ``stderr``（恒为空串）/ ``returncode``（退出码，超时为 ``-1``）/
        ``timed_out``（是否超时）/ ``cwd``（实际工作目录）/ ``argv``（实际
        执行的参数数组）/ ``truncated``（输出是否因超限被截断）。

    Raises:
        FileNotFoundError: 命令不存在。
        PermissionError: 命令不可执行。
    """
    work_dir = resolve_path(cwd) if cwd else workspace_root()
    argv = shlex.split(command)

    try:
        completed = subprocess.run(
            argv,
            shell=False,
            capture_output=True,
            text=True,
            timeout=timeout,
            cwd=str(work_dir),
        )
        stdout = _as_text(completed.stdout) + _as_text(completed.stderr)
        returncode = completed.returncode
        timed_out = False
    except subprocess.TimeoutExpired as exc:
        stdout = _as_text(exc.stdout) + _as_text(exc.stderr)
        returncode = -1
        timed_out = True

    rendered, truncated = _render_output(stdout)
    return json.dumps(
        {
            "stdout": rendered,
            "stderr": "",
            "returncode": returncode,
            "timed_out": timed_out,
            "cwd": str(work_dir),
            "argv": argv,
            "truncated": truncated,
        },
        ensure_ascii=False,
    )


# --- 内部辅助 -------------------------------------------------------------------


def _as_text(value: object) -> str:
    """把 ``subprocess`` 可能给出的 ``str`` / ``bytes`` / ``None`` 统一成文本。"""
    if value is None:
        return ""
    if isinstance(value, bytes):
        return value.decode("utf-8", errors="replace")
    return str(value)


def _render_output(text: str) -> tuple[str, bool]:
    """按行数与字节上限截断输出，返回 ``(呈现文本, 是否被截断)``。

    未被截断时**原样返回**命令的输出（含结尾换行）。被截断时保留前若干行，
    末尾附 ``[showing lines 1-<已保留> of <总数>; full output not saved]``。
    **完整输出 MUST NOT 落盘** —— 模型取剩余部分只能自行改跑更窄的命令。
    """
    lines = text.split("\n")
    if lines and lines[-1] == "":
        lines = lines[:-1]

    kept: list[str] = []
    used_bytes = 0
    truncated = False
    for raw in lines:
        if len(kept) >= MAX_OUTPUT_LINES:
            truncated = True
            break
        cost = len(raw.encode("utf-8")) + 1
        if kept and used_bytes + cost > MAX_OUTPUT_BYTES:
            truncated = True
            break
        used_bytes += cost
        kept.append(raw)

    if not truncated:
        return text, False

    kept_count = len(kept)
    kept.append(
        f"[showing lines 1-{kept_count} of {len(lines)}; full output not saved]"
    )
    return "\n".join(kept), True
